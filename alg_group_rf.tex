\begin{algorithm}[ht]
\caption{Procedure of the proposed group-based
         activity-oriented R\&F}
\label{alg:group_rf}
\begin{algorithmic}[1]
\State Determine greedy solution $S$ via serial SGS
\State Compute $\tilde{X}$ using Algorithm~\ref{alg:features}
\State Compute $\mathcal{G}=[G_1,\ldots,G_m]$
       using Algorithm~\ref{alg:grouping}
\For{$g=1$ \textbf{to} $|\mathcal{G}|$}
    \For{$i\in G_g$}
        \State $w_i=\max(2,\lfloor 0.15(LS_i-ES_i)\rfloor)$
        \State $\widetilde{W}_i=
               [\max\{ES_i,S_i-w_i\},\,
                \min\{LS_i,S_i+w_i\}]$
        \State Fix $x^B_{it}=0$
               $\forall\,t\notin\widetilde{W}_i$
    \EndFor
    \State \parbox[t]{\dimexpr\linewidth-\algorithmicindent\relax}{%
           Solve SAA-FSRCPSP subproblem with
           $x^B_{it}\in\{0,1\}$ for $i\in G_g$,
           $x^C_{jt}\in[0,1]$ for
           $j\notin G_g\cup\mathrm{fixed}$,
           and all already fixed variables}
    \If{solution $S'$ is feasible and improves makespan}
        \State Fix $x^B_{i,S'_i}:=1$,
               $x^B_{it}:=0$
               $\forall\,t\in\widetilde{W}_i\setminus\{S'_i\}$
        \State Update $S=S'$,\; $\mathrm{no\_improve}=0$
    \Else
        \State Set $\widetilde{W}_i=[ES_i,\,LS_i]$
               for all $i\in G_g$ and re-solve subproblem
        \If{solution $S'$ found}
            \State Fix $x^B_{i,S'_i}:=1$,
                   $x^B_{it}:=0$
                   $\forall\,t\in\widetilde{W}_i\setminus\{S'_i\}$
            \State Update $S=S'$
        \Else
            \State Fix $i\in G_g$ at current $S'_i$
        \EndIf
        \State $\mathrm{no\_improve}=\mathrm{no\_improve}+1$
    \EndIf
    \If{$\mathrm{no\_improve}\geq\eta$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return Solution $S$
\end{algorithmic}
\end{algorithm}
